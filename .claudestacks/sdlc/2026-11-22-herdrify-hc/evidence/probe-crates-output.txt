fn probe_pid(label: &str, pid: u32) {
    let k = std::process::Command::new("kill").args(["-0", &pid.to_string()]).output().unwrap();
    let ps = std::process::Command::new("ps").args(["-o", "pid=,stat=,comm=", "-p", &pid.to_string()]).output().unwrap();
    println!(
        "  {label}: kill -0 {pid} -> exit {:?} stderr={:?}; ps -> exit {:?} out={:?}",
        k.status.code(),
        String::from_utf8_lossy(&k.stderr).trim(),
        ps.status.code(),
        String::from_utf8_lossy(&ps.stdout).trim()
    );
}

use serde::{Deserialize, Serialize};
use serde_json::value::RawValue;
use serde_json::{Map, Value};

#[derive(Debug, Deserialize)]
struct Plain {
    id: u32,
}

#[derive(Debug, Deserialize)]
#[serde(deny_unknown_fields)]
struct Strict {
    id: u32,
}

#[derive(Debug, Serialize, Deserialize)]
struct WithExtra {
    id: u32,
    #[serde(flatten)]
    extra: Map<String, Value>,
}

#[derive(Debug, Serialize, Deserialize)]
struct Opt {
    #[serde(default)]
    name: Option<String>,
}

#[derive(Debug, Serialize, Deserialize)]
struct OptNoDefault {
    name: Option<String>,
}

#[derive(Debug, Serialize, Deserialize)]
struct OptSkip {
    #[serde(default, skip_serializing_if = "Option::is_none")]
    name: Option<String>,
}

#[derive(Debug, Serialize, Deserialize)]
struct Request {
    jsonrpc: String,
    id: Value,
    method: String,
}

#[derive(Debug, Deserialize)]
struct Response<'a> {
    #[allow(dead_code)]
    jsonrpc: String,
    id: Value,
    #[serde(borrow, default)]
    result: Option<&'a RawValue>,
    #[serde(default)]
    error: Option<Value>,
}

#[derive(Debug, Deserialize)]
struct ResponseOwned {
    id: Value,
    result: Option<Box<RawValue>>,
}

#[derive(Debug, Deserialize)]
#[serde(untagged)]
enum Incoming {
    Response { id: Value, result: Box<RawValue> },
    Error { id: Value, error: Value },
    Notification { method: String, params: Option<Box<RawValue>> },
}

#[derive(Debug, Deserialize)]
struct ResponseRequiredRaw {
    result: Box<RawValue>,
}

#[derive(Debug, Deserialize)]
#[serde(untagged)]
enum IncomingValue {
    Response { id: Value, result: Value },
    Error { id: Value, error: Value },
    Notification { method: String, params: Option<Value> },
}

#[derive(Debug, Deserialize)]
struct Hello {
    version: String,
}

fn show<T: std::fmt::Debug>(label: &str, r: Result<T, serde_json::Error>) {
    match r {
        Ok(v) => println!("{label}: Ok({v:?})"),
        Err(e) => println!("{label}: Err(\"{e}\") classify={:?} line={} column={}", e.classify(), e.line(), e.column()),
    }
}
